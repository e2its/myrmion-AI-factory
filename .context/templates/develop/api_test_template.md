# Plantilla E: API Integration Test (`tests/api/{{feature}}.api.test.{ext}`)

> **Purpose:** Template for HTTP-level API integration tests that validate actual endpoints against contracts from `contracts/`.
> These tests complement unit tests (which mock dependencies) by testing the **full HTTP request-response cycle**.
>
> **When to generate:** Feature has API endpoints defined in `design.md` or `contracts/`.
> **Source of truth:** `contracts/` (OpenAPI/GraphQL/TypeScript schemas generated by ARCH).
> **Runner:** Stack-specific — supertest (Node.js), httpx/TestClient (Python/FastAPI), net/http/httptest (Go), MockMvc (Java/Spring).
> **Incremental-slicing note.** Under `slicing_strategy: incremental`, API tests you generate are scoped to the increment currently in `BUILDING`. `increment_plan.md § 1` lists each increment's `contract_surface` — test ONLY those operations in this increment's PR. Cross-increment integration belongs in the final increment's acceptance gate.

```typescript
        // or from contracts/ request body schema
      };

      const response = await request(app)
        .post('/api/v1/{{resource}}')
        .send(payload)
        .expect(201);

      // Validate response matches contracts/ response schema
      expect(response.body).toMatchObject({
        // Fields from contracts/ response schema (types per design.md § Schema Constraints)
      });
    });
  });

  describe('GET /api/v1/{{resource}}/:id', () => {
    it('should return resource by id → 200', async () => {
      const response = await request(app)
        .get('/api/v1/{{resource}}/{{TEST_ID}}')
        .expect(200);

      expect(response.body).toHaveProperty('id');
      // Validate against contracts/ response schema
    });

    it('should return 404 for non-existent resource', async () => {
      await request(app)
        .get('/api/v1/{{resource}}/non-existent-id')
        .expect(404);
    });
  });

  // ─── Edge Cases (from test_plan.md Section 2) ───
  describe('Error Handling', () => {
    it('should reject malformed JSON → 400', async () => {
      await request(app)
        .post('/api/v1/{{resource}}')
        .set('Content-Type', 'application/json')
        .send('{ invalid json')
        .expect(400);
    });

    it('should reject missing required fields → 400/422', async () => {
      await request(app)
        .post('/api/v1/{{resource}}')
        .send({}) // Empty payload
        .expect(400); // or 422 per api-standards.instructions.md
    });
  });

  // ─── Security (from test_plan.md Section 2 + sec patterns) ───
  describe('Security', () => {
    it('should reject unauthenticated request → 401', async () => {
      await request(app)
        .get('/api/v1/{{resource}}')
        // No auth header
        .expect(401);
    });

    it('should sanitize SQL injection attempt → 400', async () => {
      await request(app)
        .post('/api/v1/{{resource}}')
        .send({ field: "' OR 1=1 --" })
        .expect(400);
    });
  });
});
```

## Pattern: Python (FastAPI + httpx/TestClient)

```python

# Contract: contracts/openapi/{{CONTRACT_SLUG}}/v1.yaml

import pytest
from httpx import AsyncClient  # or: from fastapi.testclient import TestClient
from {{APP_MODULE}} import app  # e.g., src.main

@pytest.mark.asyncio
class TestFeatureAPI:
    """API integration tests for {{FEATURE_NAME}}."""

    # ─── Happy Path ───
    async def test_create_resource_valid_payload_201(self, client: AsyncClient):
        payload = {
            # Field names from user_journey.md § 6; types/formats from design.md § Schema Constraints
        }
        response = await client.post("/api/v1/{{resource}}", json=payload)
        assert response.status_code == 201
        data = response.json()
        # Validate against contracts/ response schema
        assert "id" in data

    async def test_get_resource_by_id_200(self, client: AsyncClient):
        response = await client.get("/api/v1/{{resource}}/{{TEST_ID}}")
        assert response.status_code == 200

    async def test_get_resource_not_found_404(self, client: AsyncClient):
        response = await client.get("/api/v1/{{resource}}/non-existent")
        assert response.status_code == 404

    # ─── Edge Cases ───
    async def test_malformed_json_400(self, client: AsyncClient):
        response = await client.post(
            "/api/v1/{{resource}}",
            content="{ invalid",
            headers={"Content-Type": "application/json"},
        )
        assert response.status_code in (400, 422)

    async def test_missing_required_fields_422(self, client: AsyncClient):
        response = await client.post("/api/v1/{{resource}}", json={})
        assert response.status_code in (400, 422)

    # ─── Security ───
    async def test_unauthenticated_401(self, client: AsyncClient):
        response = await client.get("/api/v1/{{resource}}")
        # No auth header
        assert response.status_code == 401

    async def test_sql_injection_sanitized_400(self, client: AsyncClient):
        response = await client.post(
            "/api/v1/{{resource}}",
            json={"field": "' OR 1=1 --"},
        )
        assert response.status_code == 400
```

## Generation Rules

1. **One test file per API resource** (e.g., `users.api.test.ts`, `auth.api.test.ts`).
2. **Import app instance** (NOT start a server) — framework test utilities handle this.
3. **Test data** — field existence from `user_journey.md § 6 Business Fields`; types/formats/examples from `design.md § Schema Constraints` (type_format_registry) or `contracts/` request/response schemas.
4. **Happy paths** implement `test_plan.md § 2.1` TC-API-XX rows (cite the exact ID per test) → mapped to HTTP verbs + status codes.
5. **Edge/error cases** from `test_plan.md` Section 2 (Edge, Security, Negative rows with HTTP status codes).
6. **Status codes** from `contracts/` OpenAPI responses or `.claude/rules/api-standards.md`.
7. **Auth headers** if feature requires authentication (detect from `design.md` security section).
8. **File location:** `tests/api/` directory (stack-agnostic; `tests/` root from `constitution.md`) — `config/quality.json → contracts.tests_root`, the folder `contract-parity` runs.
9. **Runner command:** `./scripts/test.sh api --apply` or direct runner (`npx jest tests/api/`, `pytest tests/api/`, `go test ./tests/api/`).

## Recording the answers (EVOL-070 — contract parity)

These tests are the contract tests: they live under `contracts.tests_root` (default `tests/api/`, item 8) and send every request through one wrapper that records the answer. The loop member `contract-parity` runs them with `CONTRACT_OBSERVED` set and reads the records (`rules/contract-first-policy.md` § Contract parity). Outside that run the variable is unset and the wrapper records nothing.

One JSON line per response, in a `*.jsonl` file of that directory: `{"operation": "<operationId>", "status": <int>, "code": "<CODE>"|null}`. `code` is the error body's top-level `code` (`rules/api-standards.md`), or `null`. Every operation of the contracts is answered at least once.

```python
# Python
import json, os
def record(operation, response):
    d = os.environ.get("CONTRACT_OBSERVED")
    if d:
        body = response.json() if "json" in response.headers.get("content-type", "") else {}
        with open(os.path.join(d, f"{os.getpid()}.jsonl"), "a") as f:
            f.write(json.dumps({"operation": operation, "status": response.status_code, "code": body.get("code") if isinstance(body, dict) else None}) + "\n")
    return response
```

```typescript
// TypeScript (supertest)
import { appendFileSync } from 'fs';
export function record(operation: string, res: { status: number; body?: any }) {
  const d = process.env.CONTRACT_OBSERVED;
  if (d) appendFileSync(`${d}/${process.pid}.jsonl`, JSON.stringify({ operation, status: res.status, code: res.body?.code ?? null }) + '\n');
  return res;
}
```

```java
// Java (MockMvc result) — ObjectMapper from Jackson
static MvcResult record(String operation, MvcResult r) throws Exception {
  String d = System.getenv("CONTRACT_OBSERVED");
  if (d != null) {
    String raw = r.getResponse().getContentAsString();
    JsonNode b = raw.isEmpty() ? null : new ObjectMapper().readTree(raw);
    String code = b != null && b.hasNonNull("code") ? b.get("code").asText() : null;
    Files.writeString(Path.of(d, ProcessHandle.current().pid() + ".jsonl"), new ObjectMapper().writeValueAsString(
        Map.of("operation", operation, "status", r.getResponse().getStatus(), "code", code == null ? NullNode.getInstance() : code)) + "\n",
        StandardOpenOption.CREATE, StandardOpenOption.APPEND);
  }
  return r;
}
```

```go
// Go (httptest.ResponseRecorder)
func record(operation string, w *httptest.ResponseRecorder) *httptest.ResponseRecorder {
	if d := os.Getenv("CONTRACT_OBSERVED"); d != "" {
		var b struct{ Code *string `json:"code"` }
		_ = json.Unmarshal(w.Body.Bytes(), &b)
		line, _ := json.Marshal(map[string]any{"operation": operation, "status": w.Code, "code": b.Code})
		f, _ := os.OpenFile(filepath.Join(d, fmt.Sprintf("%d.jsonl", os.Getpid())), os.O_CREATE|os.O_APPEND|os.O_WRONLY, 0o600)
		defer f.Close()
		_, _ = f.Write(append(line, '\n'))
	}
	return w
}
```

```csharp
// C# (HttpResponseMessage from WebApplicationFactory) — one lock: xUnit runs test classes in parallel in one process
static readonly SemaphoreSlim Gate = new(1, 1);
static async Task<HttpResponseMessage> Record(string operation, HttpResponseMessage r) {
  var d = Environment.GetEnvironmentVariable("CONTRACT_OBSERVED");
  if (d is not null) {
    string? code = null;
    try { code = JsonDocument.Parse(await r.Content.ReadAsStringAsync()).RootElement.GetProperty("code").GetString(); } catch { }
    await Gate.WaitAsync();
    try { await File.AppendAllTextAsync(Path.Combine(d, $"{Environment.ProcessId}.jsonl"),
      JsonSerializer.Serialize(new { operation, status = (int)r.StatusCode, code }) + "\n"); }
    finally { Gate.Release(); }
  }
  return r;
}
```

```rust
// Rust (status + body bytes from the test client)
fn record(operation: &str, status: u16, body: &[u8]) {
    if let Ok(d) = std::env::var("CONTRACT_OBSERVED") {
        let code = serde_json::from_slice::<serde_json::Value>(body).ok().and_then(|v| v.get("code").and_then(|c| c.as_str().map(String::from)));
        let line = serde_json::json!({"operation": operation, "status": status, "code": code}).to_string() + "\n";
        use std::io::Write;
        let mut f = std::fs::OpenOptions::new().create(true).append(true).open(format!("{d}/{}.jsonl", std::process::id())).unwrap();
        f.write_all(line.as_bytes()).unwrap();
    }
}
```

## Contract Validation (Optional — Advanced)

If `.claude/rules/contract-first-policy.md` exists and `contracts/openapi/` contains OpenAPI specs:
- Use schema validation library (ajv for Node.js, jsonschema for Python) to validate response bodies match OpenAPI response schemas.
- This ensures runtime responses stay in sync with declared contracts.

```typescript

const ajv = new Ajv();
```
